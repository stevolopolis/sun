```js
const selectedSummary = selectStationRows(spellSummary, baselines, pair);
const selectedSpells = selectStationRows(extremeSpells, baselines, pair);
const pairId = String(pairs.indexOf(pair) + 1);
const pairTheme = selectedSummary[0]?.PAIR_THEME || "";

const DAY = 24 * 60 * 60 * 1000;
const windowStart = new Date(Date.UTC(startYear, 0, 1));
const windowEnd = new Date(Date.UTC(startYear, Math.round(displaySpan * 12), 1));

function parseUtcDate(value) {
  if (value instanceof Date) return value;
  if (!value) return null;
  const s = String(value).slice(0, 10);
  const parts = s.split("-");
  if (parts.length === 3 && parts[0].length === 4) {
    return new Date(Date.UTC(+parts[0], +parts[1] - 1, +parts[2]));
  }
  return new Date(value);
}

const windowDaily = dailyTemps.filter(d => {
  if (String(d.PAIR_ID) !== pairId) return false;
  const t = parseUtcDate(d.DATE);
  return t >= windowStart && t < windowEnd;
});

function getBucketDate(date, window) {
  const d = new Date(date);
  switch (window) {
    case "day": return new Date(Date.UTC(d.getUTCFullYear(), d.getUTCMonth(), d.getUTCDate()));
    case "week": {
      const day = d.getUTCDay();
      const diff = d.getUTCDate() - day + (day === 0 ? -6 : 1);
      return new Date(Date.UTC(d.getUTCFullYear(), d.getUTCMonth(), diff));
    }
    case "month": return new Date(Date.UTC(d.getUTCFullYear(), d.getUTCMonth(), 1));
    case "season": return new Date(Date.UTC(d.getUTCFullYear(), Math.floor(d.getUTCMonth() / 3) * 3, 1));
    case "year": return new Date(Date.UTC(d.getUTCFullYear(), 0, 1));
    default: return new Date(Date.UTC(d.getUTCFullYear(), d.getUTCMonth(), d.getUTCDate()));
  }
}

function aggregate(rows, window) {
  const grouped = new Map();
  rows.forEach(d => {
    const dt = parseUtcDate(d.DATE);
    if (!dt || isNaN(+dt)) return;
    const bucket = +getBucketDate(dt, window);
    if (!grouped.has(bucket)) grouped.set(bucket, []);
    grouped.get(bucket).push(+d.TEMP);
  });

  return Array.from(grouped, ([time, temps]) => ({
    date: new Date(time),
    temp: temps.reduce((a, b) => a + b, 0) / temps.length,
    count: temps.length
  })).sort((a, b) => a.date - b.date);
}

const stationNames = new Map(selectedSummary.map(d => [d.STATION_ROLE, d.STATION]));
const series = ["Station A", "Station B"].map(role => ({
  role,
  station: stationNames.get(role) || role,
  values: aggregate(windowDaily.filter(d => d.STATION_ROLE === role), averagingWindow)
}));

const spellsForWindow = selectedSpells.map(d => ({
  ...d,
  start: parseUtcDate(d.START_DATE),
  end: parseUtcDate(d.END_DATE),
  type: d.EXTREME_TYPE === "Extreme heat" ? "heat" : "cold",
  peak: +d.PEAK_TEMP,
  duration: +d.DURATION_DAYS,
  meanSeverity: +d.AVG_SEVERITY,
  recovery: d.RECOVERY_DAYS ? +d.RECOVERY_DAYS : null
}));
```

An extreme heat spell is defined as consecutive days with daily maximum temperature reaching or exceeding 90 °F (`MAX ≥ 90°F`). An extreme cold spell is defined as consecutive days with daily minimum temperature falling to or below -10 °F (`MIN ≤ -10°F`). Recovery measures the days until daily temperature returns to within 2 °F of the pre-spell 7-day baseline.

### What the visualization shows

Spell duration, persistence, and recovery dynamics reveal fundamental contrasts in thermal stability across stations:

- **Transient vs. Persistent Extremes:** Coastal, maritime, and equatorial highland locations (Arcata, San Francisco, Reykjavík, Addis Ababa, Christchurch) experience almost exclusively transient, isolated extreme episodes lasting only 1–2 consecutive days, with extreme cold below -10 °F entirely absent. Comparatively, continental interiors and deserts show persistent spells lasting weeks or months. For example, Des Moines heat waves average 3.6 days and extend up to 31 consecutive days, while desert Tucson averages 7.3 days per heat spell and has endured unbroken heat waves lasting up to 92 consecutive days (over three continuous months above 90 °F)
- **Severe Cold Concentration:** Prolonged extreme cold snaps (MIN ≤ -10 °F) are concentrated in continental subarctic and interior locations (Caribou with 163 spells totaling 541 days, and Des Moines with 63 spells totaling 130 days). In these locations, spells can stall for up to 9–13 consecutive days, driving minimum temperatures down to -24 °F to -30 °F.
- **Recovery Dynamics:** Across all pairs, temperatures typically recover back to baseline within 4 to 8 days following an extreme spell. Extreme heat spells generally break faster (averaging ~4–5 days to recover) as marine surges or cold fronts pass through, whereas severe continental cold snaps recover slower (~7–8 days)
